CLASS zcl_mdg_usmd212c_validator DEFINITION
  PUBLIC
  FINAL
  CREATE PUBLIC .

  PUBLIC SECTION.
    CLASS-METHODS:
      validate_table IMPORTING iv_tabname TYPE tabname RETURNING VALUE(rv_valid) TYPE abap_boolean,
      get_table_hash IMPORTING iv_tabname TYPE tabname RETURNING VALUE(rv_hash) TYPE char32,
      get_expected_hash IMPORTING iv_tabname TYPE tabname RETURNING VALUE(rv_hash) TYPE char32,
      check_compatibility IMPORTING iv_tabname TYPE tabname RETURNING VALUE(rv_compatible) TYPE abap_boolean.
ENDCLASS.

CLASS zcl_mdg_usmd212c_validator IMPLEMENTATION.
  METHOD validate_table.
    DATA: lt_dfies    TYPE ddfields,
          lt_expected TYPE TABLE OF string.

    rv_valid = abap_false.

    CALL FUNCTION 'DDIF_FIELDINFO_GET'
      EXPORTING
        tabname   = iv_tabname
      TABLES
        dfies_tab = lt_dfies
      EXCEPTIONS
        OTHERS    = 1.

    IF sy-subrc <> 0.
      RETURN.
    ENDIF.
    APPEND INITIAL LINE TO lt_expected ASSIGNING FIELD-SYMBOL(<fs_expected>).
    <fs_expected> = 'USMD_CREQ_TYPE'.
    APPEND INITIAL LINE TO lt_expected ASSIGNING <fs_expected>.
    <fs_expected> =  'USMD_REASON_REJ'.

    LOOP AT lt_expected INTO DATA(lv_field).
      READ TABLE lt_dfies WITH KEY fieldname = lv_field
                          TRANSPORTING NO FIELDS.
      IF sy-subrc <> 0.
        RETURN.
      ENDIF.
    ENDLOOP.

    rv_valid = abap_true.
  ENDMETHOD.

  METHOD get_table_hash.
    DATA: lt_dfies   TYPE ddfields,
          lv_string  TYPE string,
          lv_xstring TYPE xstring,
          lv_hash    TYPE hash160.

    CALL FUNCTION 'DDIF_FIELDINFO_GET'
      EXPORTING
        tabname   = iv_tabname
      TABLES
        dfies_tab = lt_dfies
      EXCEPTIONS
        OTHERS    = 1.

    IF sy-subrc <> 0.
      RETURN.
    ENDIF.

    LOOP AT lt_dfies INTO DATA(ls_dfies).
      lv_string = |{ lv_string }{ ls_dfies-fieldname }{ ls_dfies-datatype }{ ls_dfies-leng }|.
    ENDLOOP.

    TRY.
        lv_xstring = cl_abap_codepage=>convert_to( lv_string ).

        CALL FUNCTION 'CALCULATE_HASH_FOR_RAW'
          EXPORTING
            alg    = 'MD5'
            data   = lv_xstring
          IMPORTING
            hash   = rv_hash
          EXCEPTIONS
            OTHERS = 1.

        IF sy-subrc <> 0.
          rv_hash = lv_string+0(32).
        ENDIF.
      CATCH cx_root.
        rv_hash = lv_string+0(32).
    ENDTRY.
  ENDMETHOD.
  METHOD get_expected_hash.
    CASE iv_tabname.
      WHEN 'USMD212C'. rv_hash = 'UPDATE_ME_WITH_REAL_HASH'.
      WHEN OTHERS. rv_hash = get_table_hash( iv_tabname ).
    ENDCASE.
  ENDMETHOD.

  METHOD check_compatibility.
    rv_compatible = COND #( WHEN validate_table( iv_tabname ) = abap_false THEN abap_false ELSE abap_true ).
  ENDMETHOD.
ENDCLASS.
