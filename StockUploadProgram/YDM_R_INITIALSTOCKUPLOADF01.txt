FORM okcode  USING    iv_sy_ucomm.
  CASE iv_sy_ucomm.
    WHEN 'FILE_UP'.
      PERFORM get_data.
    WHEN 'OPEN'.
      PERFORM open_file_dialog.
    WHEN 'START'.
      PERFORM initial_load.
    WHEN 'LOG'.
      PERFORM display_log.
  ENDCASE.
ENDFORM.                    " okcode

*&---------------------------------------------------------------------*
*& Form submit_job
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM submit_job .


*** constants declarations..
  CONSTANTS: lc_jobname TYPE char19 VALUE 'YDM_UPL_VAL_STOCK',
             lc_u       TYPE char1 VALUE '_'.
**** Data Declarations.
  DATA: lv_jobcnt  TYPE btcjobcnt,
        lv_jobname TYPE tbtcjob-jobname,
        lv_date    TYPE sy-datum,
        lv_memid   TYPE char40.

  WRITE sy-datum TO lv_date.

*  lv_jobname = |{ lc_jobname }{ lc_u } { lv_date } { lc_u } { sy-uzeit }|.
  CONCATENATE lc_jobname lc_u lv_date   lc_u  sy-uzeit INTO lv_jobname.
  CONDENSE lv_jobname.
**** job open
  CALL FUNCTION 'JOB_OPEN'
    EXPORTING
      jobname          = lv_jobname
    IMPORTING
      jobcount         = lv_jobcnt
    EXCEPTIONS
      cant_create_job  = 1
      invalid_job_data = 2
      jobname_missing  = 3
      OTHERS           = 4.
  IF sy-subrc <> 0.
    MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
     WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
  ELSE.

**** build the SAP Memory id.
    CONCATENATE lv_memid 'ISU' '_' sy-uzeit INTO lv_memid.
***** Exporting the data to SAP Memory.
    EXPORT gt_stock[] TO DATABASE demo_indx_blob(sc) ID lv_memid.
***** Submit the job with Parameter details.
    SUBMIT ydm_upload_valid_stock_isu
            WITH p_budat = /scwm/ordim_c-mmim_budat_migration
            WITH p_lgnum = /scwm/s_t300t-lgnum
            WITH p_memid = lv_memid
            WITH p_no_erp = g_no_erp
            WITH p_postno = g_postno
            VIA JOB lv_jobname NUMBER lv_jobcnt
                AND RETURN.

****** Job close
    CALL FUNCTION 'JOB_CLOSE'
      EXPORTING
        jobcount             = lv_jobcnt
        jobname              = lv_jobname
        strtimmed            = abap_true
      EXCEPTIONS
        cant_start_immediate = 1
        invalid_startdate    = 2
        jobname_missing      = 3
        job_close_failed     = 4
        job_nosteps          = 5
        job_notex            = 6
        lock_failed          = 7
        OTHERS               = 8.
    IF sy-subrc = 0.
**** Add Successful message.
      CONCATENATE 'Job' lv_jobname 'Scheduled Successfully' INTO  gs_output-message SEPARATED BY space.
      gs_output-icon = lc_icon_01.
      APPEND gs_output TO gt_output.
    ELSE.
**** Add Error message.
      MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno "to raise exception
             WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4 INTO DATA(lv_text).
      gs_output-message    = lv_text.
      gs_output-msgty =  sy-msgty.
      IF sy-msgty = 'I'.
        gs_output-icon = lc_icon_35.
      ELSEIF  sy-msgty = 'E'.
        gs_output-icon = lc_icon_02.
      ENDIF.
      APPEND gs_output TO gt_output.

    ENDIF.
  ENDIF.


ENDFORM.
